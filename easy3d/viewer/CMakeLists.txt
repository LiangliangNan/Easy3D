set(module viewer)
set(private_dependencies)
set(public_dependencies easy3d::renderer)
set(viewer_headers)
set(viewer_sources)

if(Easy3D_ENABLE_OPENGL)
    list(APPEND private_dependencies 3rd_glfw)
    list(APPEND viewer_headers viewer.h multi_viewer.h offscreen.h)
    list(APPEND viewer_sources viewer.cpp multi_viewer.cpp snapshot.cpp offscreen.cpp)
endif()
if(Easy3D_ENABLE_VULKAN_VIEWER)
    find_package(Vulkan REQUIRED)
    if(NOT Easy3D_ENABLE_OPENGL)
        list(APPEND private_dependencies 3rd_glfw)
        list(APPEND viewer_headers viewer.h)
    endif()
    list(APPEND private_dependencies Vulkan::Vulkan)
    list(APPEND viewer_headers vulkan_viewer.h)
    list(APPEND viewer_sources vulkan_viewer.cpp)
endif()
if(Easy3D_ENABLE_VULKAN)
    list(APPEND public_dependencies easy3d::vulkan)
    list(APPEND viewer_headers vulkan_offscreen.h)
    list(APPEND viewer_sources vulkan_offscreen.cpp)
    if(NOT Easy3D_ENABLE_OPENGL)
        list(APPEND viewer_headers offscreen.h)
    endif()
endif()

add_module(viewer "${viewer_headers}" "${viewer_sources}" "${private_dependencies}" "${public_dependencies}")
if(Easy3D_ENABLE_VULKAN_VIEWER)
    target_compile_definitions(easy3d_viewer PUBLIC EASY3D_HAS_VULKAN_VIEWER)
endif()
install_module(viewer)

if(Easy3D_ENABLE_VULKAN AND Easy3D_BUILD_VULKAN_TESTS)
    add_executable(easy3d_vulkan_scene_test ../graphics/tests/scene_test.cpp)
    target_link_libraries(easy3d_vulkan_scene_test PRIVATE easy3d::viewer)
    set(validation_args)
    if(Easy3D_VULKAN_TEST_VALIDATION)
        set(validation_args --validation)
    endif()
    add_test(NAME vulkan.scene COMMAND easy3d_vulkan_scene_test ${validation_args})
    set_tests_properties(vulkan.scene PROPERTIES TIMEOUT 120)
    add_executable(easy3d_vulkan_texture_test ../graphics/tests/texture_test.cpp)
    target_link_libraries(easy3d_vulkan_texture_test PRIVATE easy3d::viewer)
    add_test(NAME vulkan.texture COMMAND easy3d_vulkan_texture_test ${validation_args})
    set_tests_properties(vulkan.texture PROPERTIES TIMEOUT 120)
    if(Easy3D_ENABLE_OPENGL)
        add_executable(easy3d_opengl_texture_test ../graphics/tests/texture_gl_test.cpp)
        target_link_libraries(easy3d_opengl_texture_test PRIVATE easy3d::viewer)
        add_test(NAME opengl.texture COMMAND easy3d_opengl_texture_test)
        set_tests_properties(opengl.texture PROPERTIES TIMEOUT 120 LABELS desktop)
    endif()
    if(Easy3D_ENABLE_VULKAN_VIEWER)
        add_executable(easy3d_vulkan_window_test ../graphics/tests/window_test.cpp)
        target_link_libraries(easy3d_vulkan_window_test PRIVATE easy3d::viewer)
        add_test(NAME vulkan.window COMMAND easy3d_vulkan_window_test ${validation_args})
        set_tests_properties(vulkan.window PROPERTIES TIMEOUT 120 LABELS "desktop")
        add_executable(easy3d_vulkan_image_viewer_test ../graphics/tests/image_viewer_test.cpp)
        target_link_libraries(easy3d_vulkan_image_viewer_test PRIVATE easy3d::viewer)
        add_test(NAME vulkan.imageviewer COMMAND easy3d_vulkan_image_viewer_test ${validation_args})
        set_tests_properties(vulkan.imageviewer PROPERTIES TIMEOUT 120 LABELS desktop)
    endif()
endif()
