cmake_minimum_required(VERSION 3.12)
project(Easy3DInstalledVulkanConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Easy3D CONFIG REQUIRED COMPONENTS viewer renderer)
if(TARGET easy3d::vulkan OR "vulkan" IN_LIST Easy3D_AVAILABLE_COMPONENTS)
    message(FATAL_ERROR "The Vulkan implementation must be part of the renderer component")
endif()
if(EXISTS "${Easy3D_INCLUDE_DIRS}/easy3d/graphics")
    message(FATAL_ERROR "The installed package still contains the retired graphics module")
endif()
if(Easy3D_ENABLE_VULKAN)
    add_executable(installed_vulkan_consumer main.cpp)
    target_link_libraries(installed_vulkan_consumer PRIVATE easy3d::viewer)
endif()

add_executable(installed_offscreen_consumer unified.cpp)
target_link_libraries(installed_offscreen_consumer PRIVATE easy3d::viewer)

add_executable(installed_viewer_consumer viewer.cpp)
target_link_libraries(installed_viewer_consumer PRIVATE easy3d::viewer)
